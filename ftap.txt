local function PlayAudio(soundId)
    pcall(function()
        local sound = Instance.new("Sound")
        sound.SoundId = "rbxassetid://" .. tostring(soundId)
        sound.Volume = 0.5
        sound.Parent = SoundService
        sound:Play()
        sound.Ended:Connect(function() sound:Destroy() end)
    end)
end

local NotificationManager = {}
function NotificationManager.Notify(title, content, duration, icon)
    pcall(function()
        WindUI:Notify({
            Title = title or "ROSEY FTAP",
            Content = content or "",
            Duration = duration or 3,
            Icon = icon or "info"
        })
    end)
end

local function Notify(title, content, duration, icon)
    NotificationManager.Notify(title, content, duration, icon)

local ConnectionManager = {}
function ConnectionManager.Add(name, connection)
    if State.Connections[name] then
        pcall(function() State.Connections[name]:Disconnect() end)
        State.Connections[name] = nil
    end
    State.Connections[name] = connection
    return connection
end

function ConnectionManager.Remove(name)
    if State.Connections[name] then
        pcall(function() State.Connections[name]:Disconnect() end)
        State.Connections[name] = nil
    end
end

function ConnectionManager.ClearAll()
    for name, conn in pairs(State.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    State.Connections = {}
end

local LoopManager = {}
function LoopManager.Start(name, func, interval)
    LoopManager.Stop(name)
    State.Loops[name] = true
    task.spawn(function()
        while State.Loops[name] do
            local success, err = pcall(func)
            if not success and State.Settings.NotifyOnError then
                Notify("Loop Error", tostring(name) .. ": " .. tostring(err), 4, "alert-triangle")
            end
            task.wait(interval or 0.1)
        end
    end)
end

function LoopManager.Stop(name)
    State.Loops[name] = false
end

function LoopManager.ClearAll()
    for name in pairs(State.Loops) do
        State.Loops[name] = false
    end
end

local PlayerManager = {}
function PlayerManager.GetPlayerList(excludeLocal)
    local list = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if not excludeLocal or player ~= LocalPlayer then
            table.insert(list, player.Name)
        end
    end
    table.sort(list)
    return list
end

function PlayerManager.RegisterDropdown(dropdown, excludeLocal)
    table.insert(State.RegisteredDropdowns, { dropdown = dropdown, excludeLocal = excludeLocal })
    pcall(function()
        dropdown:Refresh(PlayerManager.GetPlayerList(excludeLocal), true)
    end)
end

function PlayerManager.RefreshAllDropdowns()
    for _, item in ipairs(State.RegisteredDropdowns) do
        pcall(function()
            item.dropdown:Refresh(PlayerManager.GetPlayerList(item.excludeLocal), true)
        end)
    end
end

function PlayerManager.GetCharacter(player)
    player = player or LocalPlayer
    if typeof(player) == "string" then
        player = Players:FindFirstChild(player)
    end
    if player and player.Character then
        return player.Character
    end
    return nil
end

function PlayerManager.GetRootPart(player)
    local char = PlayerManager.GetCharacter(player)
    if char then
        return char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart
    end
    return nil
end

local ESPManager = {}
function ESPManager.CreateHighlight(player)
    if player == LocalPlayer or not player.Character then return end
    if State.ESPObjects[player] then
        pcall(function() State.ESPObjects[player]:Destroy() end)
    end

    local highlight = Instance.new("Highlight")
    highlight.Name = "ROSEY_ESP_" .. player.Name
    highlight.Adornee = player.Character
    highlight.FillColor = State.Settings.ESPColor
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Enabled = State.Settings.ESPEnabled
    highlight.Parent = player.Character
    State.ESPObjects[player] = highlight
end

function ESPManager.Refresh()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if State.Settings.ESPEnabled and player.Character then
                ESPManager.CreateHighlight(player)
            elseif not State.Settings.ESPEnabled and State.ESPObjects[player] then
                pcall(function() State.ESPObjects[player]:Destroy() end)
                State.ESPObjects[player] = nil
            end
        end
    end
end

function ESPManager.Cleanup()
    for _, obj in pairs(State.ESPObjects) do
        pcall(function() obj:Destroy() end)
    end
    State.ESPObjects = {}
end

function ConfigManager.Load()
    local success, err = pcall(function()
        if readfile and isfile and isfile("RoseyFTAP_Config.json") then
            local data = HttpService:JSONDecode(readfile("RoseyFTAP_Config.json"))

                State.Settings[k] = v

            if State.Settings.NotifyOnConfig then
                Notify("Config", "Configuration loaded successfully.", 3, "check")

        else
            Notify("Config", "No configuration file found.", 3, "info")
        end
    end)

        Notify("Config Error", "Failed to load: " .. tostring(err), 3, "alert-triangle")
    end
end

    Desc = "Launches released GrabParts with multiplier velocity in camera look direction",
    Default = State.Settings.StrengthActive,

        State.Settings.StrengthActive = val
        if val then
            ConnectionManager.Add("StrengthConnection", Workspace.ChildAdded:Connect(function(model)
                if model.Name == "GrabParts" and State.Settings.StrengthActive then
                    local grabPart = model:FindFirstChild("GrabPart")
                    local weld = grabPart and grabPart:FindFirstChild("WeldConstraint")
                    local impulsePart = weld and weld.Part1

                    if impulsePart then
                        local velocityObj = Instance.new("BodyVelocity")
                        velocityObj.MaxForce = Vector3.zero
                        velocityObj.Parent = impulsePart

                        -- Support Mobile Context Action Buttons
                        local mobileBtn = nil
                        pcall(function()
                            for _, v in pairs(LocalPlayer.PlayerGui.ContextActionGui:GetDescendants()) do
                                if v:IsA("ImageLabel") and v.Image == "http://www.roblox.com/asset/?id=9603678090" then
                                    mobileBtn = v.Parent
                                    break
                                end
                            end
                        end)

                        local function FireImpulse()
                            velocityObj.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                            velocityObj.Velocity = Camera.CFrame.LookVector * State.Settings.LaunchStrength
                        end

                        local mConn1, mConn2
                        if mobileBtn then
                            mConn1 = mobileBtn.MouseButton1Up:Connect(FireImpulse)
                            mConn2 = mobileBtn.MouseButton1Down:Connect(FireImpulse)
                        end

                        local pConn
                        pConn = model:GetPropertyChangedSignal("Parent"):Connect(function()
                            if not model.Parent then
                                if UserInputService:GetLastInputType() == Enum.UserInputType.MouseButton2 then
                                    FireImpulse()
                                    Debris:AddItem(velocityObj, 1)
                                else
                                    velocityObj:Destroy()
                                end
                                if pConn then pConn:Disconnect() end
                                if mConn1 then mConn1:Disconnect() end
                                if mConn2 then mConn2:Disconnect() end
                            end
                        end)
                    end
                end
            end))
            Notify("Combat", "Super Strength Engine enabled", 2, "check")
        else
            ConnectionManager.Remove("StrengthConnection")
            Notify("Combat", "Super Strength Engine disabled", 2, "x")
        end
    end

Tabs.Combat:Toggle({
    Title = "Camera Aim Assist",
    Desc = "Interpolates camera toward closest target within FOV",
    Default = false,
    Callback = function(val)
        aimFovCircle.Visible = val
        if val then
            ConnectionManager.Add("AimbotStep", RunService.RenderStepped:Connect(function()
                aimFovCircle.Position = Camera.ViewportSize / 2
                aimFovCircle.Radius = State.Settings.AimFov

                local closestPart = nil
                local shortestDist = State.Settings.AimFov

                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and player.Character then
                        local targetPart = player.Character:FindFirstChild(State.Settings.AimPart)
                        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                        if targetPart and humanoid and humanoid.Health > 0 then
                            local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                            if onScreen then
                                local mouseDist = (Vector2.new(screenPos.X, screenPos.Y) - (Camera.ViewportSize / 2)).Magnitude
                                if mouseDist < shortestDist then
                                    shortestDist = mouseDist
                                    closestPart = targetPart
                                end
                            end
                        end
                    end
                end

                if closestPart then
                    Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
                end
            end))
        else
            ConnectionManager.Remove("AimbotStep")
            aimFovCircle.Visible = false
        end
    end
})

Tabs.Auras:Section({ Title = "Super Ring Parts Tornado (V7)" })

local function IsValidRingPart(part)
    if part:IsA("BasePart") and not part.Anchored and part:IsDescendantOf(Workspace) then
        if part.Parent == LocalPlayer.Character or part:IsDescendantOf(LocalPlayer.Character) then
            return false
        end
        part.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
        part.CanCollide = false
        return true
    end
    return false
end

local function IndexRingParts()
    State.RingTrackedParts = {}
    for _, part in ipairs(Workspace:GetDescendants()) do
        if IsValidRingPart(part) then
            table.insert(State.RingTrackedParts, part)
        end
    end
end

Tabs.Auras:Toggle({
    Title = "Enable Super Ring Parts Vortex",
    Desc = "Draws all unanchored physics objects into a rotating vortex around you",
    Default = State.Settings.RingPartsEnabled,
    Callback = function(val)
        State.Settings.RingPartsEnabled = val
        PlayAudio("12221967")
        if val then
            IndexRingParts()
            ConnectionManager.Add("RingDescAdded", Workspace.DescendantAdded:Connect(function(part)
                if IsValidRingPart(part) and not table.find(State.RingTrackedParts, part) then
                    table.insert(State.RingTrackedParts, part)
                end
            end))

            ConnectionManager.Add("RingDescRemoved", Workspace.DescendantRemoving:Connect(function(part)
                local idx = table.find(State.RingTrackedParts, part)
                if idx then table.remove(State.RingTrackedParts, idx) end
            end))

            ConnectionManager.Add("RingHeartbeat", RunService.Heartbeat:Connect(function()
                if not State.Settings.RingPartsEnabled then return end
                local root = PlayerManager.GetRootPart()
                if root then
                    local center = root.Position
                    for _, part in ipairs(State.RingTrackedParts) do
                        if part.Parent and not part.Anchored then
                            local pos = part.Position
                            local dist = (Vector3.new(pos.X, center.Y, pos.Z) - center).Magnitude
                            local angle = math.atan2(pos.Z - center.Z, pos.X - center.X)
                            local newAngle = angle + math.rad(State.Settings.RingRotationSpeed)
                            local targetPos = Vector3.new(
                                center.X + math.cos(newAngle) * math.min(State.Settings.RingRadius, dist),
                                center.Y + (State.Settings.RingHeight * (math.abs(math.sin((pos.Y - center.Y) / State.Settings.RingHeight)))),
                                center.Z + math.sin(newAngle) * math.min(State.Settings.RingRadius, dist)
                            )
                            part.Velocity = (targetPos - part.Position).Unit * State.Settings.RingAttractionStrength
                        end
                    end
                end
            end))
            Notify("Auras", "Super Ring Parts Tornado Active!", 3, "orbit")
        else
            ConnectionManager.Remove("RingDescAdded")
            ConnectionManager.Remove("RingDescRemoved")
            ConnectionManager.Remove("RingHeartbeat")
            State.RingTrackedParts = {}
            Notify("Auras", "Super Ring Parts Disabled", 2, "x")
        end
    end
})
